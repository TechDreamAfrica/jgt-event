Supabase SQL Setup Script

Run this SQL in your Supabase SQL Editor to create the registrations table with Row Level Security (RLS) configured for anonymous registrations: 

-- 1. Create registrations table
CREATE TABLE IF NOT EXISTS public.registrations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    full_name TEXT NOT NULL,
    email TEXT NOT NULL UNIQUE,
    phone TEXT NOT NULL,
    church_organization TEXT,
    attendees INTEGER NOT NULL DEFAULT 1,
    message TEXT,
    updates_consent BOOLEAN NOT NULL DEFAULT true,
    registration_reference TEXT UNIQUE NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    status TEXT DEFAULT 'registered'
);

-- 2. Enable Row Level Security
ALTER TABLE public.registrations ENABLE ROW LEVEL SECURITY;

-- 3. Policy: Allow anonymous users to INSERT new registrations
CREATE POLICY "Allow public insert" 
ON public.registrations 
FOR INSERT 
TO anon 
WITH CHECK (true);

-- 4. Policy: Restrict SELECT to authenticated service role / admins only
-- Anonymous users cannot read, update, or delete registrations.
