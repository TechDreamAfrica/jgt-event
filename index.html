Supabase SQL Setup Script

Run this SQL in your Supabase SQL Editor to create the registrations table with complete Row Level Security (RLS) for registration and admin operations: 

-- 1. Create registrations table
CREATE TABLE IF NOT EXISTS public.registrations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    full_name TEXT NOT NULL,
    email TEXT NOT NULL UNIQUE,
    phone TEXT NOT NULL,
    church_organization TEXT,
    attendees INTEGER NOT NULL DEFAULT 1,
    message TEXT,
    updates_consent BOOLEAN NOT NULL DEFAULT true,
    registration_reference TEXT UNIQUE NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    status TEXT DEFAULT 'registered'
);

-- 2. Enable Row Level Security
ALTER TABLE public.registrations ENABLE ROW LEVEL SECURITY;

-- 3. Policy: Allow public insert for registrations
CREATE POLICY "Allow public insert" 
ON public.registrations 
FOR INSERT 
TO anon 
WITH CHECK (true);

-- 4. Policy: Allow public select for registrations and counts
CREATE POLICY "Allow public select" 
ON public.registrations 
FOR SELECT 
TO anon 
USING (true);

-- 5. Policy: Allow public update (e.g. check-in status changes)
CREATE POLICY "Allow public update" 
ON public.registrations 
FOR UPDATE 
TO anon 
USING (true);

-- 6. Policy: Allow public delete (for admin deletion)
CREATE POLICY "Allow public delete" 
ON public.registrations 
FOR DELETE 
TO anon 
USING (true);
